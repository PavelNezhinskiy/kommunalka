РЕЗЕРВНАЯ КОПИЯ
Копия в Google Sheets уходит автоматически раз в неделю при открытии приложения. Кнопками можно отправить или скачать прямо сейчас.
Каждое показание пишется на лист «Из приложения», полная копия — на листы «Копия · …»
Как настроить (один раз):
1. Откройте таблицу → Расширения → Apps Script
2. Удалите весь код, вставьте код ниже → сохраните (Ctrl+S)
3. Если скрипт уже разворачивался: «Начать развертывание» → Управление развертываниями → карандаш → Версия: Новая версия → Развернуть. URL не изменится.
4. Если впервые: «Начать развертывание» → Новое развертывание → тип Веб-приложение · Доступ: Все → скопировать URL в поле выше
5. Проверка: откройте URL скрипта с ?ping=1 на конце — должно ответить v2. Если пусто или ошибка — развёрнута старая версия
function doGet(e) { try { // проверка версии: откройте URL скрипта с ?ping=1 — должно ответить v2 if (e && e.parameter && e.parameter.ping) return out('v2'); const ss = SpreadsheetApp.getActiveSpreadsheet(); const rows = JSON.parse(decodeURIComponent(e.parameter.data || '[]')); const arr = Array.isArray(rows) ? rows : [rows]; if (e.parameter.sheet === 'tariffs') { let sh = ss.getSheetByName('Тарифы (приложение)'); if (!sh) { sh = ss.insertSheet('Тарифы (приложение)'); sh.appendRow(['Изменено','С даты','Тип','Фикс. плата (€/год)','Цена за ед. (€)','Действие']); sh.getRange(1,1,1,6).setFontWeight('bold').setBackground('#fef3c7'); } arr.forEach(r => sh.appendRow([new Date(), r.effective_from, r.type, r.fixed_cost, r.rate, r.action||''])); } else { let sh = ss.getSheetByName('Из приложения'); if (!sh) { sh = ss.insertSheet('Из приложения'); sh.appendRow(['Внесено','Дата','Тип','Комната','Показание','Δ потр.','Стоимость €']); sh.getRange(1,1,1,7).setFontWeight('bold').setBackground('#e8f0fe'); } arr.forEach(r => sh.appendRow([new Date(), r.date, r.type, r.room||'', r.value, r.delta||'', r.cost||''])); } return out('ok'); } catch(err) { return out('err:' + err.message); } } // Полная резервная копия базы — вызывается приложением function doPost(e) { try { const body = JSON.parse(e.postData.contents); if (body.kind !== 'backup') return out('err:unknown kind'); const ss = SpreadsheetApp.getActiveSpreadsheet(); let tables = 0, total = 0; Object.keys(body.tables).forEach(function(name) { const rows = body.tables[name]; if (!rows || !rows.length) return; const title = 'Копия · ' + name; let sh = ss.getSheetByName(title) || ss.insertSheet(title); sh.clear(); const cols = Object.keys(rows[0]); const data = [cols].concat(rows.map(function(r) { return cols.map(function(c) { return r[c] === null ? '' : String(r[c]); }); })); sh.getRange(1, 1, data.length, cols.length).setValues(data); sh.getRange(1, 1, 1, cols.length).setFontWeight('bold').setBackground('#dcfce7'); sh.setFrozenRows(1); tables++; total += rows.length; }); let log = ss.getSheetByName('Копии · журнал'); if (!log) { log = ss.insertSheet('Копии · журнал'); log.appendRow(['Когда','Таблиц','Строк всего']); log.getRange(1,1,1,3).setFontWeight('bold').setBackground('#dcfce7'); } log.appendRow([new Date(), tables, total]); return out('ok:' + total); } catch(err) { return out('err:' + err.message); } } function out(s) { return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.TEXT); }
